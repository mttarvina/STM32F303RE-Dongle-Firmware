cmake_minimum_required(VERSION 3.22)

# Set the project name
set(CMAKE_PROJECT_NAME STM32F303RE-Dongle-Firmware)

# Force absolute path for toolchain file if not provided on command line
if(NOT CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/cmake/gcc-arm-none-eabi.cmake" CACHE FILEPATH "ARM Toolchain")
endif()

# Setup C compiler standard settings
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

# Define default build type if not provided
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Debug")
endif()

# Enable compilation database generation for Zed / clangd
set(CMAKE_EXPORT_COMPILE_COMMANDS TRUE)

# ------------------------------------------------------------------------------
# Core Project Initialization
# ------------------------------------------------------------------------------
project(${CMAKE_PROJECT_NAME} C ASM)
message("Build type: " ${CMAKE_BUILD_TYPE})

# 1. CREATE THE TARGET FIRST (CubeMX subdirectory expects this target to exist)
add_executable(${CMAKE_PROJECT_NAME})

# 2. Include STM32CubeMX autogenerated build system
add_subdirectory(cmake/stm32cubemx)

# ==============================================================================
# Custom User Libraries (TarvsLib)
# ==============================================================================

# Automatically collect all C sources inside TarvsLib/Src
file(GLOB_RECURSE TARVS_LIB_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/TarvsLib/Src/*.c"
)

# Add custom library source files to the project target
target_sources(${CMAKE_PROJECT_NAME} PRIVATE
    ${TARVS_LIB_SOURCES}
)

# Add custom library include header paths
target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/TarvsLib/Inc
)

# Link directories setup
target_link_directories(${CMAKE_PROJECT_NAME} PRIVATE
    # Add user-defined library search paths here
)

# Custom preprocessor flags/macros
target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE
    # Add user-defined symbols here
)

# Remove wrong libob.a library dependency when using cpp files
list(REMOVE_ITEM CMAKE_C_IMPLICIT_LINK_LIBRARIES ob)

# ------------------------------------------------------------------------------
# Target Linking
# ------------------------------------------------------------------------------
target_link_libraries(${CMAKE_PROJECT_NAME} stm32cubemx)

# ------------------------------------------------------------------------------
# Post-Build Steps: Generate .bin/.hex and copy all release artifacts to repo
# ------------------------------------------------------------------------------
set(RELEASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Bin")

add_custom_command(
    TARGET ${CMAKE_PROJECT_NAME} POST_BUILD
    # 1. Generate .bin and .hex files from .elf
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${CMAKE_PROJECT_NAME}> ${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_PROJECT_NAME}.bin
    COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:${CMAKE_PROJECT_NAME}> ${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_PROJECT_NAME}.hex

    # 2. Ensure release directory exists in the repo root
    COMMAND ${CMAKE_COMMAND} -E make_directory ${RELEASE_DIR}

    # 3. Copy .elf, .bin, and .hex files to release directory
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:${CMAKE_PROJECT_NAME}> ${RELEASE_DIR}/${CMAKE_PROJECT_NAME}.elf
    COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_PROJECT_NAME}.bin ${RELEASE_DIR}/${CMAKE_PROJECT_NAME}.bin
    COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_PROJECT_NAME}.hex ${RELEASE_DIR}/${CMAKE_PROJECT_NAME}.hex

    COMMENT "Copying ELF, BIN, and HEX release artifacts to ${RELEASE_DIR}..."
)
